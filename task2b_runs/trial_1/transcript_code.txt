for L in labels:
    a=data[L]; x=a[:,0]-0; y=a[:,1]-0
    th=np.unwrap(np.arctan2(y,x))
    r=np.hypot(x,y)
    total = abs(th[-1]-th[0])/(2*np.pi)
    # bin r by angle mod 2pi
    # Fourier: fit r = sum over k of amplitude via mean of r*exp(i k th)
    amps=[]
    for k in range(1,13):
        c=np.mean(r*np.exp(-1j*k*th))
        amps.append(abs(c))
    amps=np.array(amps)
    best=np.argmax(amps)+1
    print(L, "turns=%.2f"%total, "best k=",best, " ".join("%d:%.2f"%(i+1,v) for i,v in enumerate(amps)))
