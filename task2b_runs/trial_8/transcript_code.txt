import csv, ast, numpy as np
p=r"C:\Users\xan37\Documents\GitHub\Judge\task2b_data.csv"
rows=list(csv.reader(open(p)))
labels=rows[0][1:]
data={l:np.array([ast.literal_eval(r[i+1]) for r in rows[1:]],dtype=float) for i,l in enumerate(labels)}
for l in labels:
    a=data[l]; x,y=a[:,0],a[:,1]
    th=np.arctan2(y,x); r=np.hypot(x,y)
    # fourier of r vs theta
    res=[]
    for k in range(1,10):
        c=np.mean(r*np.exp(-1j*k*th))
        res.append(round(abs(c),3))
    # also unwrapped angle total turning and count of radius local maxima
    thu=np.unwrap(th)
    turns=(thu[-1]-thu[0])/(2*np.pi)
    # local maxima of r
    lm=np.sum((r[1:-1]>r[:-2])&(r[1:-1]>r[2:]))
    print(l,res,round(turns,2),lm, round(r.mean(),2), round(r.max(),2))
