import csv, ast, numpy as np
path=r"C:\Users\xan37\Documents\GitHub\Judge\task2b_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
t=np.array([float(r[0]) for r in rows[1:]])
data={l:np.array([ast.literal_eval(r[i+1]) for r in rows[1:]]) for i,l in enumerate(labels)}
for l in labels:
    p=data[l]; x,y=p[:,0],p[:,1]
    th=np.arctan2(y,x); r=np.hypot(x,y)
    # local maxima of r
    idx=[i for i in range(1,len(r)-1) if r[i]>r[i-1] and r[i]>r[i+1] and r[i]>10]
    ang=th[idx]
    # test N by resultant of N*ang
    res=[(N,abs(np.mean(np.exp(1j*N*ang)))) for N in range(2,9)]
    best=sorted(res,key=lambda z:-z[1])[:3]
    print(l, len(idx), [(N,round(v,3)) for N,v in best])
