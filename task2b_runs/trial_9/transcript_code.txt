import csv, ast, numpy as np

path = r"C:\Users\xan37\Documents\GitHub\Judge\task2b_data.csv"
rows = list(csv.reader(open(path)))
labels = rows[0][1:]
data = {l: [] for l in labels}
t=[]
for r in rows[1:]:
    t.append(float(r[0]))
    for l,c in zip(labels,r[1:]):
        data[l].append(ast.literal_eval(c))
t=np.array(t)
for l in labels: data[l]=np.array(data[l])

for l in labels:
    x,y = data[l][:,0], data[l][:,1]
    r = np.hypot(x,y)
    th = np.unwrap(np.arctan2(y,x))
    total = abs(th[-1]-th[0])/(2*np.pi)
    # count local maxima of r
    peaks = np.sum((r[1:-1]>r[:-2])&(r[1:-1]>r[2:]))
    # spectral: interpolate r as function of theta
    thn = th if th[-1]>th[0] else -th
    # resample
    N=4096
    g = np.linspace(thn[0], thn[-1], N)
    ri = np.interp(g, thn, r)
    ri = ri - ri.mean()
    F = np.abs(np.fft.rfft(ri*np.hanning(N)))
    freqs = np.fft.rfftfreq(N, d=(g[1]-g[0]))*2*np.pi  # cycles per radian *2pi = per revolution
    k = freqs[np.argmax(F)]
    top = np.argsort(F)[::-1][:5]
    print(l, "revs %.2f"%total, "peaks", peaks, "peaks/rev %.2f"%(peaks/total), "domk %.2f"%k,
          [round(freqs[i],2) for i in top])
