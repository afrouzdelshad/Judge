import csv, ast, numpy as np
path=r"C:\Users\xan37\Documents\GitHub\Judge\task2b_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
D={l:[] for l in labels}
for r in rows[1:]:
    for l,c in zip(labels,r[1:]):
        D[l].append(ast.literal_eval(c))
for l in labels:
    a=np.array(D[l]); x,y=a[:,0],a[:,1]
    r=np.hypot(x,y); th=np.arctan2(y,x)
    # local maxima of r
    idx=[i for i in range(1,len(r)-1) if r[i]>r[i-1] and r[i]>r[i+1] and r[i]>np.percentile(r,60)]
    ang=th[idx]
    res=[]
    for N in range(2,10):
        res.append((N, round(abs(np.mean(np.exp(1j*N*ang))),3)))
    best=max(res,key=lambda z:z[1])
    print(l,len(idx),best,res)
