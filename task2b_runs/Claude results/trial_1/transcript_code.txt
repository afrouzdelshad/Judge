import csv, ast, numpy as np
path=r"C:\Users\xan37\Documents\GitHub\Judge\task2b_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
T=len(rows)-1
data={l:np.zeros((T,2)) for l in labels}
for i,r in enumerate(rows[1:]):
    for j,l in enumerate(labels):
        data[l][i]=ast.literal_eval(r[j+1])

for l in labels:
    d=data[l]
    x=d[:,0]-d[:,0].mean(); y=d[:,1]-d[:,1].mean()
    th=np.unwrap(np.arctan2(y,x)); r=np.hypot(x,y)
    dth=th[-1]-th[0]
    # local maxima of r
    m=(r[1:-1]>r[:-2])&(r[1:-1]>r[2:])
    nmax=m.sum()
    # spectral: r as function of theta, interpolate uniform
    if dth<0: th=-th
    order=np.argsort(th)
    thi=np.linspace(th.min(),th.max(),4096)
    ri=np.interp(thi,th[order],r[order])
    F=np.abs(np.fft.rfft(ri-ri.mean()))
    freqs=np.fft.rfftfreq(4096, d=(thi[-1]-thi[0])/4095)*2*np.pi  # cycles per radian *2pi
    k=freqs[np.argmax(F)]
    print(l, round(dth/(2*np.pi),2), nmax, round(nmax/abs(dth/(2*np.pi)),2), round(k,2))
