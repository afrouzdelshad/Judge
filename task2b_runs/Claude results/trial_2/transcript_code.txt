labels=rows[0][1:]
data={l:[] for l in labels}

for r in rows[1:]:
    t.append(float(r[0]))
    for l,c in zip(labels,r[1:]):
        data[l].append(ast.literal_eval(c))
t=np.array(t); 
for l in labels: data[l]=np.array(data[l])

def analyze(xy):
    x,y=xy[:,0],xy[:,1]
    xc,yc=x.mean(),y.mean()
    th=np.unwrap(np.arctan2(y-yc,x-xc))
    r=np.hypot(x-xc,y-yc)
    dtheta=th[-1]-th[0]
    revs=abs(dtheta)/(2*np.pi)
    # count radial maxima
    pk=0
    idx=[]
    for i in range(1,len(r)-1):
        if r[i]>r[i-1] and r[i]>=r[i+1]:
            pk+=1; idx.append(i)
    # fft of r
    rr=r-r.mean()
    f=np.fft.rfft(rr*np.hanning(len(rr)))
    freqs=np.fft.rfftfreq(len(rr), t[1]-t[0])
    fr=freqs[np.argmax(np.abs(f))]
    # fft of angle-derivative dominant
    return revs, pk, pk/revs, fr, fr/(revs/(t[-1]-t[0]))
print("lab revs peaks peaks/rev rfreq ratio")
for l in labels:
    a=analyze(data[l])
    print(l, " ".join(f"{v:.3f}" for v in a))
