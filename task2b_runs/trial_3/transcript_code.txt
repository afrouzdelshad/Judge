out=[]
for l in labels:
    a=data[l]
    x=a[:,0]-a[:,0].mean(); y=a[:,1]-a[:,1].mean()
    th=np.arctan2(y,x); r=np.hypot(x,y)
    # binned r profile
    nb=120
    idx=((th+np.pi)/(2*np.pi)*nb).astype(int)%nb
    prof=np.array([r[idx==b].mean() if (idx==b).any() else np.nan for b in range(nb)])
    prof=np.where(np.isnan(prof),np.nanmean(prof),prof)
    F=np.abs(np.fft.rfft(prof-prof.mean()))[:15]
    kbest=int(np.argmax(F))
    # radial spectrum in time
    rr=r-r.mean()
    P=np.abs(np.fft.rfft(rr*np.hanning(len(rr))))**2
    freqs=np.fft.rfftfreq(len(rr),d=t[1]-t[0])

    thu=np.unwrap(th); om=(thu[-1]-thu[0])/(t[-1]-t[0])/(2*np.pi)
