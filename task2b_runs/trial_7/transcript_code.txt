import csv, ast, numpy as np

path = r"C:\Users\xan37\Documents\GitHub\Judge\task2b_data.csv"
with open(path) as f:
    rd = list(csv.reader(f))
labels = rd[0][1:]
t = np.array([float(r[0]) for r in rd[1:]])
data = {lab: np.array([ast.literal_eval(r[i+1]) for r in rd[1:]]) for i,lab in enumerate(labels)}

for lab in labels:
    xy = data[lab]
    x,y = xy[:,0]-0, xy[:,1]-0
    r = np.hypot(x,y); th = np.arctan2(y,x)
    # local maxima of r
    idx = [i for i in range(1,len(r)-1) if r[i]>r[i-1] and r[i]>=r[i+1]]
    angs = th[idx]
    # fourier of r vs theta: compute |sum r*exp(i k th)| for k=1..12
    amps=[]
    for k in range(1,13):
        amps.append(abs(np.sum((r-r.mean())*np.exp(1j*k*th)))/len(r))
    best = int(np.argmax(amps))+1
    # also max-angle clustering: count via k that maximizes |sum exp(i k angs)|
    amps2=[abs(np.sum(np.exp(1j*k*angs)))/len(angs) for k in range(1,13)]
    best2=int(np.argmax(amps2))+1
    print(lab, "nmax",len(idx), "kR",best, [round(a,2) for a in amps], "kA",best2,[round(a,2) for a in amps2])
