import csv, ast, numpy as np

path = r"C:\Users\xan37\Documents\GitHub\Judge\task2b_data.csv"
rows = list(csv.reader(open(path)))
labels = rows[0][1:]
data = {lab: np.array([ast.literal_eval(r[j+1]) for r in rows[1:]]) for j,lab in enumerate(labels)}

def analyze(lab, cx, cy):
    a=data[lab]; X=a[:,0]-cx; Y=a[:,1]-cy
    th=np.unwrap(np.arctan2(Y,X)); r=np.hypot(X,Y)
    if th[-1]<th[0]: th=-th
    turns=(th[-1]-th[0])/(2*np.pi)
    grid=np.linspace(th[0],th[-1],8000)
    rg=np.interp(grid,th,r)
    # smooth
    k=np.ones(15)/15; rgs=np.convolve(rg,k,mode='same')[20:-20]
    F=np.abs(np.fft.rfft(rg-rg.mean()))
    freqs=np.fft.rfftfreq(len(grid),d=(grid[1]-grid[0])/(2*np.pi))
    idx=np.argsort(F)[-6:][::-1]
    return turns, list(zip(np.round(freqs[idx],2), np.round(F[idx]/F.max(),2)))

print("center = global mean of all points")
allpts=np.vstack([data[l] for l in labels])
gx,gy=allpts[:,0].mean(),allpts[:,1].mean()
print(gx,gy)
for lab in labels:
    a=data[lab]
    print(lab, "mean",np.round(a.mean(0),2), "range x",np.round(a[:,0].min(),1),np.round(a[:,0].max(),1),
          "y",np.round(a[:,1].min(),1),np.round(a[:,1].max(),1))
    print("   selfcenter:",analyze(lab,a[:,0].mean(),a[:,1].mean()))
