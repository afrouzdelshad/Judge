import csv, ast, numpy as np

path = r"C:\Users\xan37\Documents\GitHub\Judge\task3b_data.csv"
with open(path) as f:
    rd = csv.reader(f); hdr=next(rd); rows=list(rd)
labels=hdr[1:]
data={lab: np.array([ast.literal_eval(r[i+1]) for r in rows]) for i,lab in enumerate(labels)}

for lab in labels:
    p=data[lab]
    x=p[:,0]; y=p[:,1]
    cx,cy=x.mean(),y.mean()
    th=np.arctan2(y-cy,x-cx)
    r=np.hypot(x-cx,y-cy)
    # angular histogram weighted, look for k-fold symmetry
    res=[]
    for k in range(1,13):
        c=np.abs(np.mean(np.exp(1j*k*th)))
        # also weighted by radius (lobe tips far out)
        w=(r>np.percentile(r,80))
        c2=np.abs(np.mean(np.exp(1j*k*th[w])))
        res.append((k,round(c,3),round(c2,3)))
    best=max(res,key=lambda z:z[2])
    print(lab, "x range %.1f %.1f  y %.1f %.1f"%(x.min(),x.max(),y.min(),y.max()), "best k=",best, [f"{k}:{c2}" for k,c,c2 in res])
