path = r"C:\Users\xan37\Documents\GitHub\Judge\task3b_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
data={lab:np.array([ast.literal_eval(r[i+1]) for r in rows[1:]]) for i,lab in enumerate(labels)}
for lab in labels:
    p=data[lab]; x=p[:,0]-p[:,0].mean(); y=p[:,1]-p[:,1].mean()
    r=np.hypot(x,y); th=np.arctan2(y,x)
    idx=[i for i in range(1,len(r)-1) if r[i]>r[i-1] and r[i]>=r[i+1] and r[i]>np.percentile(r,60)]
    ang=th[idx]
    scores=[]
    for N in range(1,13):
        c=np.abs(np.mean(np.exp(1j*N*ang)))
        scores.append((round(c,3),N))
    scores.sort(reverse=True)
    print(lab, len(idx), scores[:4])
