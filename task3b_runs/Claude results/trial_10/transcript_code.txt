import csv, ast, numpy as np
path=r"C:\Users\xan37\Documents\GitHub\Judge\task3b_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
t=np.array([float(r[0]) for r in rows[1:]])
data={l:np.array([ast.literal_eval(r[i+1]) for r in rows[1:]]) for i,l in enumerate(labels)}

for l in labels:
    a=data[l]; x,y=a[:,0],a[:,1]
    r=np.hypot(x,y); th=np.arctan2(y,x)
    # local maxima of r
    idx=[i for i in range(1,len(r)-1) if r[i]>r[i-1] and r[i]>r[i+1] and r[i]>np.percentile(r,60)]
    angs=th[idx]
    # unwrapped angle total winding
    tot=np.abs(np.unwrap(th)[-1]-np.unwrap(th)[0])/(2*np.pi)
    # FFT of r vs angle: estimate petal count k by correlating r with cos(k*th)
    scores=[]
    for k in range(1,13):
        c=np.hypot(np.mean(r*np.cos(k*th)),np.mean(r*np.sin(k*th)))
        scores.append(round(c,3))
    print(l, "npeaks=%d"%len(idx), "wind=%.1f"%tot, "best_k=%d"%(1+int(np.argmax(scores))), scores)
