import csv, ast, numpy as np
path = r"C:\Users\xan37\Documents\GitHub\Judge\task3b_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
data={lab:np.array([ast.literal_eval(r[i+1]) for r in rows[1:]]) for i,lab in enumerate(labels)}

for lab in labels:
    a=data[lab]; x,y=a[:,0],a[:,1]
    r=np.hypot(x,y); th=np.arctan2(y,x)
    m=r>np.percentile(r,60)
    h,_=np.histogram(th[m],bins=360,range=(-np.pi,np.pi))
    # fourier magnitudes of angular histogram
    F=np.abs(np.fft.rfft(h-h.mean()))[:15]
    order=np.argsort(F)[::-1][:4]
    print(lab, "top k:",order, np.round(F[order]/F.max(),2))
