import csv, ast, numpy as np
path = r"C:\Users\xan37\Documents\GitHub\Judge\task3b_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
data={lab:np.array([ast.literal_eval(r[i+1]) for r in rows[1:]]) for i,lab in enumerate(labels)}

for lab in labels:
    p=data[lab]; x,y=p[:,0],p[:,1]
    th=np.arctan2(y,x); r=np.hypot(x,y)
    # binned mean radius vs angle
    nb=360
    idx=((th+np.pi)/(2*np.pi)*nb).astype(int)%nb
    prof=np.zeros(nb); cnt=np.zeros(nb)
    np.add.at(prof,idx,r); np.add.at(cnt,idx,1)
    prof=prof/np.maximum(cnt,1)
    prof=prof-prof.mean()
    F=np.abs(np.fft.rfft(prof))
    top=np.argsort(F[1:15])[::-1][:4]+1
    print(lab, "harmonics:", [(k,round(F[k],1)) for k in top], "rmin/max", round(r.min(),1), round(r.max(),1))
