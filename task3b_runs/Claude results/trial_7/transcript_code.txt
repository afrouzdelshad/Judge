import csv, ast, numpy as np

path = r"C:\Users\xan37\Documents\GitHub\Judge\task3b_data.csv"
with open(path) as f:
    rows = list(csv.reader(f))
labels = rows[0][1:]
data = {L: [] for L in labels}
for r in rows[1:]:
    for L, cell in zip(labels, r[1:]):
        data[L].append(ast.literal_eval(cell))
for L in labels:
    data[L] = np.array(data[L])

for L in labels:
    xy = data[L]
    x, y = xy[:,0], xy[:,1]
    r = np.hypot(x,y); th = np.arctan2(y,x)
    # use outer points
    thr = np.percentile(r, 70)
    m = r > thr
    scores = []
    for n in range(1,13):
        s = np.abs(np.mean(np.exp(1j*n*th[m])))
        scores.append(round(s,3))
    best = int(np.argmax(scores))+1
    print(L, best, scores)
