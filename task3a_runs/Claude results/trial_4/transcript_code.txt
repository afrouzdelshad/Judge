import csv, ast
import numpy as np
path=r"C:\Users\xan37\Documents\GitHub\Judge\task3a_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
data={lab:np.array([ast.literal_eval(r[j+1]) for r in rows[1:]]) for j,lab in enumerate(labels)}

ics={}
for lab in labels:
    key=tuple(np.round(data[lab][0],3))
    ics.setdefault(key,[]).append(lab)
print(ics)

for lab in labels:
    a=data[lab]
    x,y=a[:,0],a[:,1]
    r=np.hypot(x,y); th=np.arctan2(y,x)
    # local maxima of r
    idx=[i for i in range(1,len(r)-1) if r[i]>r[i-1] and r[i]>r[i+1] and r[i]>10]
    ang=np.sort(np.degrees(th[idx])%360)
    h,_=np.histogram(ang,bins=36,range=(0,360))
    # unwrapped angle total winding
    tw=np.abs(np.unwrap(th)[-1]-th[0])/(2*np.pi)
    print(lab, len(idx), "wind=%.2f"%tw, "maxperwind=%.2f"%(len(idx)/tw), "".join('%d'%min(x,9) for x in h))
