    for j,l in enumerate(labels):

for l in labels:
    d=data[l]; x,y=d[:,0],d[:,1]
    cx,cy=x.mean(),y.mean()
    r=np.hypot(x-cx,y-cy); th=np.arctan2(y-cy,x-cx)
    # local maxima of r above threshold
    idx=[i for i in range(1,T-1) if r[i]>r[i-1] and r[i]>=r[i+1] and r[i]>np.percentile(r,80)]
    angs=th[idx]
    # harmonic strength
    res=[]
    for k in range(1,13):
        res.append((k, round(abs(np.exp(1j*k*angs).mean()),3)))
    best=max(res,key=lambda z:z[1])
    print(l, len(idx), best, res)
