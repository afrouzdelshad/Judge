import csv, ast, numpy as np
p=r"C:\Users\xan37\Documents\GitHub\Judge\task3a_data.csv"
rows=list(csv.reader(open(p)))
hdr=rows[0]; labels=hdr[1:]
T=len(rows)-1
data={l:np.zeros((T,2)) for l in labels}
t=np.zeros(T)
for i,r in enumerate(rows[1:]):
    t[i]=float(r[0])
    for j,l in enumerate(labels):
        x,y=ast.literal_eval(r[j+1]); data[l][i]=(x,y)
print(T,labels)
for l in labels:
    xy=data[l]
    x,y=xy[:,0],xy[:,1]
    cx,cy=x.mean(),y.mean()
    th=np.unwrap(np.arctan2(y-cy,x-cx))
    rr=np.hypot(x-cx,y-cy)
    total=abs(th[-1]-th[0])/(2*np.pi)
    # count radial maxima
    m=0
    for i in range(1,T-1):
        if rr[i]>rr[i-1] and rr[i]>=rr[i+1]: m+=1
    # FFT of r resampled in theta
    ths=th if th[-1]>th[0] else -th
    idx=np.argsort(ths)
    grid=np.linspace(ths[idx][0],ths[idx][-1],4096)
    rg=np.interp(grid,ths[idx],rr[idx])
    rg=rg-rg.mean()
    F=np.abs(np.fft.rfft(rg))
    freqs=np.fft.rfftfreq(len(grid),d=(grid[-1]-grid[0])/(len(grid)-1))*2*np.pi
    top=np.argsort(F)[::-1][:5]
    print(l,"rev=%.2f"%total,"maxima=%d"%m,"ratio=%.2f"%(m/total if total>0 else 0),
          [(round(freqs[k],2),round(F[k])) for k in top])
