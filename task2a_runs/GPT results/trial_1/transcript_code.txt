import csv, ast
import numpy as np

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]

data = {
    lab: np.array([ast.literal_eval(r[j+1]) for r in rows[1:]], float)
    for j, lab in enumerate(labels)

print("shape", len(t), len(labels), "dt", np.median(np.diff(t)))
print("labels", labels)

# Analyze radius as a function of polar angle about several plausible centers:
# raw mean, coordinate median, and circle-ish center. Estimate dominant number
# of radial oscillations per revolution by linear regression phase frequencies,
# angular binning, and FFT after interpolation against unwrapped angle.
for lab in labels:
    xy = data[lab]
    # robust central location and arithmetic mean
    centers = [xy.mean(axis=0), np.median(xy, axis=0)]
    results = []
    for c in centers:
        z = (xy[:,0]-c[0]) + 1j*(xy[:,1]-c[1])
        th = np.unwrap(np.angle(z))
        r = np.abs(z)
        totalrev = abs(th[-1]-th[0])/(2*np.pi)
        monot = abs(np.sum(np.diff(th)))/(np.sum(abs(np.diff(th)))+1e-15)
        # resample radius on monotonic unwrapped angle if direction is mostly fixed
        direction = 1 if th[-1] >= th[0] else -1
        q = direction*th
        # enforce sorting; average approximately via interpolation after sort
        order = np.argsort(q)
        qs, rs = q[order], r[order]
        uq = np.linspace(np.percentile(qs,1), np.percentile(qs,99), 4096)
        ur = np.interp(uq, qs, rs)
        ur -= np.polyval(np.polyfit(uq, ur, 2), uq)
        ft = np.abs(np.fft.rfft(ur*np.hanning(len(ur))))
        freq = np.fft.rfftfreq(len(ur), d=(uq[1]-uq[0]))*2*np.pi # cycles per 2pi
        mask = (freq >= 1) & (freq <= 20)
        inds = np.where(mask)[0]
        top = inds[np.argsort(ft[inds])[-5:][::-1]]
        peaks = [(round(freq[k],2), round(ft[k]/(ft[top[0]]+1e-15),2)) for k in top]
        results.append((round(totalrev,2), round(monot,3), peaks))
    # time-domain dominant frequencies of x,y,r, useful ratio orbital/lobe
    dt = np.median(np.diff(t))
    def topsig(v, n=4):
        v = v - np.mean(v)
        a = np.abs(np.fft.rfft(v*np.hanning(len(v))))
        fr = np.fft.rfftfreq(len(v),dt)
        ii = np.arange(1,len(fr))
        top = ii[np.argsort(a[ii])[-n:][::-1]]
        return [(round(fr[k],5), round(a[k]/a[top[0]],2)) for k in top]
    c = xy.mean(0)
    rr = np.linalg.norm(xy-c,axis=1)
    print(lab, "polar", results, "Fx", topsig(xy[:,0]), "Fy", topsig(xy[:,1]), "Fr", topsig(rr))
