import csv, ast, numpy as np

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {}
for j, lab in enumerate(labels, 1):
    data[lab] = np.array([ast.literal_eval(r[j]) for r in rows[1:]], float)

print("npoints", len(t), "labels", labels)
for lab in labels:
    z = data[lab][:,0] + 1j*data[lab][:,1]
    # Estimate center as mean and analyze radial modulation versus unwrapped angle.
    c = np.mean(z)
    zz = z-c
    th = np.unwrap(np.angle(zz))
    r = np.abs(zz)
    turns = abs((th[-1]-th[0])/(2*np.pi))
    # Spectrum of radius after interpolation to uniform angle (handles variable speed).
    if th[-1] < th[0]:
        th, r = -th, r
    order = np.argsort(th)
    ths, rs = th[order], r[order]
    ug = np.linspace(ths[0], ths[-1], len(ths))
    rg = np.interp(ug, ths, rs)
    rg -= np.polyval(np.polyfit(ug, rg, min(3, len(rg)-1)), ug)
    spec = np.abs(np.fft.rfft(rg))
    freq = np.fft.rfftfreq(len(rg), d=(ug[1]-ug[0])/(2*np.pi)) # cycles per 2pi angle
    inds = np.argsort(spec[1:])[::-1][:8]+1
    peaks = [(round(float(freq[i]),3), round(float(spec[i]/(spec[inds[0]]+1e-12)),3)) for i in inds]
    # Count extrema/significant radial peaks directly
    dr = np.diff(r)
    maxima = np.sum((dr[:-1]>0)&(dr[1:]<=0))
    print(lab, "center", np.round([c.real,c.imag],3), "turns", round(turns,3),
          "rmean/std", round(r.mean(),3), round(r.std(),3), "maxima", maxima, "peaks", peaks[:5])
