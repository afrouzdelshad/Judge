import csv, ast, numpy as np
from scipy.signal import find_peaks
from scipy.fft import rfft, rfftfreq

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {lab: np.array([ast.literal_eval(r[j+1]) for r in rows[1:]], float)
        for j, lab in enumerate(labels)}

print("labels", labels, "n", len(t), "time", t[0], t[-1])

for lab in labels:
    z = data[lab][:,0] + 1j*data[lab][:,1]
    # Estimate center as mean and examine polar radius versus unwrapped angle.
    c = z.mean()
    zz = z-c
    th = np.unwrap(np.angle(zz))
    r = np.abs(zz)
    turns = (th[-1]-th[0])/(2*np.pi)
    
    # Number of radial maxima per angular revolution from detrended phase regression.
    direction = 1 if turns >= 0 else -1
    phi = direction*th
    # Harmonic coherence of radius against polar angle for candidate lobe counts.
    rr = r-r.mean()
    scores = []
    for k in range(1, 21):
        a = abs(np.mean(rr*np.exp(-1j*k*phi))) / (np.std(rr)+1e-15)
        scores.append(a)
    top = np.argsort(scores)[::-1][:5] + 1
    
    # Also count peaks, retaining reasonably prominent maxima.
    prom = max(np.std(r)*0.25, 1e-10)
    peaks, _ = find_peaks(r, prominence=prom, distance=max(1, len(r)//500))
    lobes_per_turn = len(peaks)/max(abs(turns), 1e-9)
    print(lab, "center", tuple(np.round([c.real,c.imag],3)),
          "turns", round(turns,3), "rCV", round(np.std(r)/np.mean(r),3),
          "peaks", len(peaks), "pk/turn", round(lobes_per_turn,2),
          "harmonics", [(int(k), round(scores[k-1],3)) for k in top])
