import csv, ast
import numpy as np

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"

with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {
    lab: np.array([ast.literal_eval(r[j+1]) for r in rows[1:]], dtype=float)
    for j, lab in enumerate(labels)
}

print("samples:", len(t), "labels:", labels, "time range:", (t[0], t[-1]))

for lab in labels:
    xy = data[lab]
    z = xy[:,0] + 1j*xy[:,1]
    center = np.mean(z)
    zc = z - center
    theta = np.unwrap(np.angle(zc))
    r = np.abs(zc)
    
    # Fit radius as Fourier series in unwrapped polar angle; lobe number should
    # produce strong periodic content at that angular harmonic.
    harmonics = np.arange(1, 21)
    powers = []
    for k in harmonics:
        c = np.mean((r-r.mean()) * np.exp(-1j*k*theta))
        powers.append(abs(c))
    powers = np.array(powers)
    top = harmonics[np.argsort(powers)[::-1][:6]]
    
    # Also analyze geometric turning angle and radial extrema.
    d = np.diff(z)
    tangent = np.unwrap(np.angle(d))
    total_orbits = abs(theta[-1]-theta[0])/(2*np.pi)
    total_turn = abs(tangent[-1]-tangent[0])/(2*np.pi)
    
    # Spectrum against angular coordinate after interpolation to uniform theta,
    # if sufficiently monotone.
    direction = np.sign(np.median(np.diff(theta))) or 1
    th = direction*theta
    frac_good = np.mean(np.diff(th) > 0)
    spec_top = []
    if frac_good > .8:
        order = np.argsort(th)
        thu, idx = np.unique(th[order], return_index=True)
        ru = r[order][idx]
        grid = np.linspace(thu[0], thu[-1], max(1024, len(r)))
        rg = np.interp(grid, thu, ru)
        ft = abs(np.fft.rfft((rg-rg.mean())*np.hanning(len(rg))))
        # frequency in cycles per 2pi angle
        freq = np.fft.rfftfreq(len(rg), d=(grid[1]-grid[0]))*2*np.pi
        inds = np.argsort(ft[1:])[::-1][:8]+1
        spec_top = [(round(freq[i],3), round(ft[i]/ft[inds[0]],3)) for i in inds]
    
    print(lab,
          "center", tuple(np.round([center.real, center.imag],3)),
          "orbits", round(total_orbits,2),
          "turn", round(total_turn,2),
          "monotone", round(frac_good,3),
          "polar_top", [(int(k), round(powers[k-1]/powers.max(),3)) for k in top],
          "uniform_top", spec_top)
