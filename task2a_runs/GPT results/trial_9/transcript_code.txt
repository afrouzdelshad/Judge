import csv, ast, numpy as np

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {}
for j, lab in enumerate(labels, 1):
    data[lab] = np.array([ast.literal_eval(r[j]) for r in rows[1:]], float)

print("labels:", labels)
print("samples:", len(t), "time:", (t[0], t[-1]), "dt median:", np.median(np.diff(t)))

def analyze(xy):
    z = xy[:,0] + 1j*xy[:,1]
    # Centers to compare: raw origin and coordinate mean
    out = []
    for cname, c in [("origin", 0j), ("mean", np.mean(z))]:
        w = z-c
        th = np.unwrap(np.angle(w))
        r = np.abs(w)
        turns = (th[-1]-th[0])/(2*np.pi)
        # regress radius on angular harmonics; lobe count maximizes explained variance
        detr = r - np.mean(r)
        scores = []
        for n in range(1,21):
            A=np.column_stack([np.cos(n*th),np.sin(n*th),np.ones(len(th)),th])
            fit=A@np.linalg.lstsq(A,r,rcond=None)[0]
            scores.append(1-np.var(r-fit)/np.var(r))
        # Fourier over angle via complex phase coherence, insensitive to nonuniform angular sampling
        coh=[abs(np.mean(detr*np.exp(-1j*n*th)))/np.std(r) for n in range(1,21)]
        out.append((cname,c,turns,np.argmax(scores)+1,max(scores),np.argmax(coh)+1,max(coh),scores,coh))
    return out

for lab in labels:
    a=analyze(data[lab])
    vals=[]
    for cname,c,turns,ns,ss,nc,cc,_,_ in a:
        vals.append(f"{cname}:turn={turns:.2f} fitN={ns}({ss:.3f}) cohN={nc}({cc:.3f})")
    print(lab, " | ".join(vals))
