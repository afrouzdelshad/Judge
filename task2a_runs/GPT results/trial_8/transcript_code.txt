import csv, ast, numpy as np
from scipy.signal import find_peaks, periodogram
from scipy.ndimage import gaussian_filter1d

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"

with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {}
for j, lab in enumerate(labels, 1):
    data[lab] = np.array([ast.literal_eval(r[j]) for r in rows[1:]], float)

print("npoints", len(t), "labels", labels)
print("dt", np.median(np.diff(t)), "duration", t[-1]-t[0])

def harmonic_scores(xy, center):
    z = (xy[:,0]-center[0]) + 1j*(xy[:,1]-center[1])
    th = np.unwrap(np.angle(z))
    r = np.abs(z)
    direction = 1 if th[-1] >= th[0] else -1
    u = direction*th
    # Regression/Fourier power of radius against polar angle.
    scores = []
    for n in range(1, 21):
        X = np.column_stack([np.ones(len(r)), np.cos(n*u), np.sin(n*u)])
        coef, *_ = np.linalg.lstsq(X, r, rcond=None)
        pred = X @ coef
        scores.append(1 - np.var(r-pred)/np.var(r))
    return u, r, np.array(scores)

for lab in labels:
    xy = data[lab]
    # Origin is the stated central axis; also compare trajectory mean.
    centers = [np.array([0.,0.]), xy.mean(axis=0)]
    outputs = []
    for center in centers:
        u, r, hs = harmonic_scores(xy, center)
        # radial extrema per net angular revolution
        rs = gaussian_filter1d(r, max(1, len(r)/2000))
        prom = max(np.std(rs)*0.15, 1e-12)
        peaks, _ = find_peaks(rs, prominence=prom, distance=max(2, len(r)//1000))
        turns = (u[-1]-u[0])/(2*np.pi)
        ratio = len(peaks)/turns if abs(turns) > .1 else np.nan
        top = np.argsort(hs)[-6:][::-1] + 1
        outputs.append((turns, ratio, [(int(k), round(hs[k-1],4)) for k in top]))
    
    # Time-frequency peaks for complex coordinate and radius from origin.
    dt = np.median(np.diff(t))
    z = (xy[:,0]-xy[:,0].mean()) + 1j*(xy[:,1]-xy[:,1].mean())
    n = len(z)
    freqs = np.fft.fftfreq(n, dt)
    p = np.abs(np.fft.fft(z*np.hanning(n)))**2
    inds = np.where(freqs > 0)[0]
    topz = inds[np.argsort(p[inds])[-6:][::-1]]
    r0 = np.linalg.norm(xy, axis=1)
    fr, pr = periodogram(r0, fs=1/dt, window="hann")
    ir = np.argsort(pr[1:])[-5:][::-1] + 1
    print(
        lab,
        "range", np.round(np.ptp(xy,axis=0),3),
        "mean", np.round(xy.mean(axis=0),3),
        "| origin(turns,peak/turn,harm)", outputs[0],
        "| mean-center", outputs[1],
        "| zfreq", [(round(freqs[k],5), round(p[k]/p[topz[0]],3)) for k in topz],
        "| rfreq", [(round(fr[k],5), round(pr[k]/pr[ir[0]],3)) for k in ir]
    )

# Pairwise feature-based outlier check: normalized polar-radius angular Fourier
# coefficients (magnitudes), plus temporal spectral peak ratios.
features = []
for lab in labels:

    u, r, hs = harmonic_scores(xy, np.array([0.,0.]))
    # Direct angular Fourier correlations, robust to phase.
    rr = (r-r.mean())/(r.std()+1e-12)
    cor = []
    for k in range(1,31):
        cor.append(abs(np.mean(rr*np.exp(-1j*k*u)))**2)
    features.append(np.r_[hs, cor])
F = np.array(features)
F = (F-np.median(F,axis=0))/(np.median(np.abs(F-np.median(F,axis=0)),axis=0)+1e-8)
D = np.sqrt(((F[:,None,:]-F[None,:,:])**2).mean(axis=2))
print("median feature distance by label:")
print(sorted(zip(labels, np.median(D,axis=1)), key=lambda q:q[1], reverse=True))
