import csv, ast, numpy as np

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))
labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {lab: np.array([ast.literal_eval(r[j+1]) for r in rows[1:]], float)
        for j, lab in enumerate(labels)}

print("nrows", len(t), "labels", labels, "dt", np.median(np.diff(t)))

# For each trajectory, center robustly, transform to polar coordinates, and estimate
# radial cycles per angular revolution (the likely lobe count) in several ways.
for lab in labels:
    z = data[lab][:,0] + 1j*data[lab][:,1]
    # candidate centers: mean and midpoint of coordinate extrema
    centers = [
        np.mean(z),
        (np.min(z.real)+np.max(z.real))/2 + 1j*(np.min(z.imag)+np.max(z.imag))/2,
        np.median(z.real)+1j*np.median(z.imag)
    ]
    outs = []
    for c in centers:
        w = z-c
        th = np.unwrap(np.angle(w))
        r = np.abs(w)
        # linear relation theta versus sample; total turns and radial FFT peak
        direction = np.sign(np.polyfit(np.arange(len(th)), th, 1)[0])
        u = direction*th
        turns = (u[-1]-u[0])/(2*np.pi)
        # regress radial variation against harmonics of angle, score N=1..15
        scores=[]
        for n in range(1,16):
            X=np.column_stack([np.ones(len(r)), np.cos(n*u), np.sin(n*u),
                               np.cos(2*n*u), np.sin(2*n*u)])
            pred=X@np.linalg.lstsq(X,r,rcond=None)[0]
            scores.append(1-np.var(r-pred)/np.var(r))
        # number of local radial maxima, after mild smoothing
        k=max(3, (len(r)//500)|1)
        rs=np.convolve(r,np.ones(k)/k,mode='same')
        peaks=np.sum((rs[1:-1]>rs[:-2])&(rs[1:-1]>=rs[2:]))
        outs.append((turns, int(np.argmax(scores)+1), max(scores), peaks, peaks/max(abs(turns),1e-9)))
    print(lab, ["turns=%.2f bestN=%d R2=%.3f peaks=%d ratio=%.2f"%o for o in outs])
