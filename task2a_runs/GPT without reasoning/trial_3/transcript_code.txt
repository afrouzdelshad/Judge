import csv, ast
import numpy as np

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {lab: np.array([ast.literal_eval(r[j+1]) for r in rows[1:]], float)
        for j, lab in enumerate(labels)}

print("npoints", len(t), "labels", labels, "t range", (t[0], t[-1]))
for lab in labels:
    z = data[lab]
    x, y = z[:,0], z[:,1]
    # Center by mean and inspect polar behavior
    xc, yc = x-x.mean(), y-y.mean()
    th = np.unwrap(np.arctan2(yc, xc))
    r = np.hypot(xc, yc)
    turns = (th[-1]-th[0])/(2*np.pi)
    
    # Dominant temporal frequencies of x+iy and radius
    q = (xc + 1j*yc) - np.mean(xc + 1j*yc)
    fq = np.abs(np.fft.fft(q))[1:len(q)//2]
    fr = np.abs(np.fft.rfft(r-r.mean()))[1:]
    iq = np.argmax(fq)+1
    ir = np.argmax(fr)+1
    
    # Fit Fourier harmonics of radius versus angle; report strongest 1..12
    scores = []
    for k in range(1, 13):
        A = np.column_stack([np.ones(len(r)), np.cos(k*th), np.sin(k*th)])
        pred = A @ np.linalg.lstsq(A, r, rcond=None)[0]
        scores.append(1 - np.var(r-pred)/np.var(r))
    top = np.argsort(scores)[::-1][:4] + 1
    print(lab, "turns", round(turns,2), "fft_xy", iq, "fft_r", ir,
          "ratio", round(ir/iq,2), "polar tops",
          [(int(k), round(scores[k-1],3)) for k in top],
          "ranges", tuple(np.round(np.ptp(z,axis=0),2)))
