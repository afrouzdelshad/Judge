import csv, ast
import numpy as np

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {lab: np.array([ast.literal_eval(r[j+1]) for r in rows[1:]], float)
        for j, lab in enumerate(labels)}

print("samples", len(t), "labels", labels, "dt", np.median(np.diff(t)))

for lab in labels:
    z = data[lab][:,0] + 1j*data[lab][:,1]
    # Estimate central axis in the projected plane by the coordinate mean.
    c = np.mean(z)
    w = z-c
    theta = np.unwrap(np.angle(w))
    turns = abs(theta[-1]-theta[0])/(2*np.pi)
    r = np.abs(w)
    
    # Dominant temporal frequencies of complex position and radius.
    fc = np.abs(np.fft.rfft((w.real-np.mean(w.real))))**2 + np.abs(np.fft.rfft((w.imag-np.mean(w.imag))))**2
    fr = np.abs(np.fft.rfft(r-np.mean(r)))**2
    fc[0] = fr[0] = 0
    topc = np.argsort(fc)[-6:][::-1]
    topr = np.argsort(fr)[-6:][::-1]
    
    # Fit polar radius against integer angular harmonics; report explained
    # variance for harmonics 1..12 (using sin and cos together).
    scores = []
    rv = np.var(r)
    for n in range(1,13):
        X = np.column_stack([np.ones(len(theta)), np.cos(n*theta), np.sin(n*theta)])
        pred = X @ np.linalg.lstsq(X, r, rcond=None)[0]
        scores.append(1 - np.var(r-pred)/rv if rv else 0)
    
    print(lab, "turns", round(turns,2),
          "complex bins", topc[:4].tolist(),
          "rad bins", topr[:4].tolist(),
          "ratio", round(topr[0]/topc[0],3) if topc[0] else None,
          "polar best", int(np.argmax(scores)+1), round(max(scores),3),
          "scores", [round(x,2) for x in scores])
