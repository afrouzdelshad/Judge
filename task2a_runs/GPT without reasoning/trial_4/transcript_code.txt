import csv, ast, numpy as np
from scipy.signal import find_peaks
from scipy.interpolate import interp1d

path = "/Users/adelshad3/Documents/GitHub/Judge/task2a_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {}
for j, lab in enumerate(labels, 1):
    data[lab] = np.array([ast.literal_eval(r[j]) for r in rows[1:]], float)

print("samples", len(t), "labels", labels, "time", (t[0], t[-1]))
print("label  turns  angle_range  radial_peaks  dominant_r(theta) harmonics(power)  angular_speed_cv")
for lab in labels:
    p = data[lab]
    # Robust center: use mean and also test median; trajectories are expected to orbit central axis.
    c = p.mean(axis=0)
    z = (p[:,0]-c[0]) + 1j*(p[:,1]-c[1])
    th = np.unwrap(np.angle(z))
    r = np.abs(z)
    direction = np.sign(np.median(np.diff(th))) or 1
    q = direction * th
    # Keep longest broadly monotone angular representation by sorting, then bin/interpolate.
    order = np.argsort(q)
    qs, inds = np.unique(q[order], return_index=True)
    rs = r[order][inds]
    grid = np.linspace(qs[0], qs[-1], 4096)
    rg = np.interp(grid, qs, rs)
    rg -= np.mean(rg)
    # Spectrum in cycles per 2pi of angle.
    freq = np.fft.rfftfreq(len(grid), d=(grid[1]-grid[0])) * 2*np.pi
    power = np.abs(np.fft.rfft(rg))**2
    # Aggregate power around integer angular harmonics 1..20.
    hp = []
    for k in range(1,21):
        mask = np.abs(freq-k) < max(.3, 2*np.pi/(qs[-1]-qs[0]))
        hp.append(power[mask].sum() if mask.any() else 0)
    tops = np.argsort(hp)[::-1][:5] + 1
    # Direct radial peak count after light smoothing.
    ker = np.ones(9)/9
    sm = np.convolve(r, ker, mode='same')
    peaks, _ = find_peaks(sm, prominence=0.08*np.ptp(sm), distance=max(2,len(r)//100))
    dth = np.diff(th)
    cv = np.std(dth)/max(abs(np.mean(dth)),1e-12)
    print(f"{lab:>2} {abs(th[-1]-th[0])/(2*np.pi):7.2f} {np.ptp(th):10.2f} {len(peaks):12d} "
          f"{tops[0]:8d} " + " ".join(f"{k}:{hp[k-1]/max(hp):.2f}" for k in tops) + f"  {cv:8.2f}")

# Shape symmetry via normalized complex moments E[(z/|z|)^k], and radial-angular covariance.
print("\nAngular/radial harmonic regression R2 by k=1..16 (top 4):")
for lab in labels:
    p=data[lab]; c=p.mean(0)
    z=(p[:,0]-c[0])+1j*(p[:,1]-c[1]); a=np.angle(z); r=np.abs(z)
    scores=[]
    for k in range(1,17):
        X=np.column_stack([np.ones(len(r)),np.cos(k*a),np.sin(k*a)])
        pred=X@np.linalg.lstsq(X,r,rcond=None)[0]
        scores.append(1-np.sum((r-pred)**2)/np.sum((r-r.mean())**2))
    ix=np.argsort(scores)[::-1][:4]
    print(lab, " ".join(f"{i+1}:{scores[i]:.3f}" for i in ix))
