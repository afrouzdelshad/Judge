import csv, ast, numpy as np
p=r"D:\Documents\GitHub\LLMJudge\task2_data.csv"
rows=list(csv.reader(open(p)))
hdr=rows[0]; labels=hdr[1:]
data={l:[] for l in labels}
t=[]
for r in rows[1:]:
    t.append(float(r[0]))
    for i,l in enumerate(labels):
        data[l].append(ast.literal_eval(r[i+1]))
t=np.array(t)
for l in labels:
    a=np.array(data[l]); x,y=a[:,0],a[:,1]
    cx,cy=x.mean(),y.mean()
    th=np.unwrap(np.arctan2(y-cy,x-cx))
    r=np.hypot(x-cx,y-cy)
    total=abs(th[-1]-th[0])/(2*np.pi)
    # resample r vs theta
    res=[]
    for k in range(2,10):
        c=np.abs(np.sum(r*np.exp(-1j*k*th)))/len(r)
        res.append((round(c,3),k))
    best=max(res)
    print(l, round(total,2), best, res)
