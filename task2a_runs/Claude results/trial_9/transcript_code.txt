import csv, ast, numpy as np
path=r"D:\Documents\GitHub\LLMJudge\task2_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
data={l:[] for l in labels}
for r in rows[1:]:
    for l,c in zip(labels,r[1:]):
        data[l].append(ast.literal_eval(c))

def smooth(v,w=7):
    k=np.ones(w)/w
    return np.convolve(v,k,mode='same')

for l in labels:
    a=np.array(data[l]); x,y=a[:,0],a[:,1]
    r=np.hypot(x,y)
    rs=smooth(r,9)
    idx=[i for i in range(2,len(rs)-2) if rs[i]==max(rs[i-2:i+3]) and rs[i]>r.mean()]
    # dedupe close indices
    keep=[]
    for i in idx:
        if not keep or i-keep[-1]>3: keep.append(i)
    ang=np.arctan2(y[keep],x[keep])
    res=[]
    for N in range(2,10):
        z=np.exp(1j*N*ang)
        res.append((abs(z.mean()),N))
    res.sort(reverse=True)
    print(l, len(keep), [(N,round(v,2)) for v,N in res[:3]])
