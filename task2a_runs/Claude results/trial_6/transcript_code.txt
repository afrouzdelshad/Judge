import csv, ast, numpy as np
path=r"D:\Documents\GitHub\LLMJudge\task2_data.csv"
rows=list(csv.reader(open(path)))
hdr=rows[0]; labels=hdr[1:]
data={l:[] for l in labels}
t=[]
for r in rows[1:]:
    t.append(float(r[0]))
    for i,l in enumerate(labels):
        data[l].append(ast.literal_eval(r[i+1]))
t=np.array(t)
for l in labels:
    a=np.array(data[l]); x,y=a[:,0],a[:,1]
    th=np.unwrap(np.arctan2(y,x)); r=np.hypot(x,y)
    rr=r-r.mean()
    scores=[]
    for k in range(1,10):
        c=np.abs(np.mean(rr*np.exp(1j*k*th)))
        scores.append(c)
    best=int(np.argmax(scores))+1
    print(l, best, " ".join(f"{s:.3f}" for s in scores), f"turns={(th[-1]-th[0])/2/np.pi:.2f}")
