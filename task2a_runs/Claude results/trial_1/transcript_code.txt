path = r"D:\Documents\GitHub\LLMJudge\task2_data.csv"

labels = rows[0][1:]
data = {l: [] for l in labels}
for r in rows[1:]:
    for i,l in enumerate(labels):
        data[l].append(ast.literal_eval(r[i+1]))

for l in labels:
    a = np.array(data[l]); x,y = a[:,0], a[:,1]
    cx, cy = x.mean(), y.mean()
    th = np.arctan2(y-cy, x-cx); rr = np.hypot(x-cx, y-cy)
    # mean radius as function of angle
    nb = 72
    idx = ((th+np.pi)/(2*np.pi)*nb).astype(int)%nb
    prof = np.array([rr[idx==k].mean() if np.any(idx==k) else np.nan for k in range(nb)])
    prof = np.where(np.isnan(prof), np.nanmean(prof), prof)
    F = np.abs(np.fft.rfft(prof-prof.mean()))
    top = np.argsort(F[1:11])[::-1][:3]+1
    # also count histogram of angles
    hist = np.array([np.sum(idx==k) for k in range(nb)])
    Fh = np.abs(np.fft.rfft(hist-hist.mean()))
    toph = np.argsort(Fh[1:11])[::-1][:3]+1
    print(l, "rprof", top, np.round(F[1:9],1), "| hist", toph)
