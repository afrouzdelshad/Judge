import csv, ast, numpy as np
f=open(r"D:\Documents\GitHub\LLMJudge\task2_data.csv")
r=csv.reader(f); hdr=next(r); labels=hdr[1:]; rows=list(r)
t=np.array([float(x[0]) for x in rows])
data={l:np.array([ast.literal_eval(x[i+1]) for x in rows]) for i,l in enumerate(labels)}

for l in labels:
    p=data[l]
    x=p[:,0]-p[:,0].mean(); y=p[:,1]-p[:,1].mean()
    th=np.unwrap(np.arctan2(y,x))
    rad=np.hypot(x,y)
    # resample rad vs theta uniformly
    if th[-1]<th[0]: th=-th
    idx=np.argsort(th)
    ths=th[idx]; rs=rad[idx]
    grid=np.linspace(ths[0],ths[-1],4000)
    ri=np.interp(grid,ths,rs)
    ri=ri-ri.mean()
    F=np.abs(np.fft.rfft(ri))
    span=(grid[-1]-grid[0])/(2*np.pi)  # revolutions
    freqs=np.arange(len(F))/span
    # find peak among freq 1..12
    mask=(freqs>1.5)&(freqs<12)
    k=np.argmax(F[mask])
    best=freqs[mask][k]
    order=np.argsort(F[mask])[::-1][:3]
    print(l, round(best,2), [(round(freqs[mask][o],2), round(F[mask][o])) for o in order])
