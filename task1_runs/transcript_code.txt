def circ_smooth(y, win):
    if win <= 1:
        return y.copy()
    k = np.ones(win)/win
    pad = win
    yy = np.r_[y[-pad:], y, y[:pad]]
    return np.convolve(yy, k, mode='same')[pad:-pad]

def count_peaks_circular(y, prominence_frac=0.20, min_sep=8):
    n = len(y)
    ymax, ymin = np.nanmax(y), np.nanmin(y)
    prom = (ymax-ymin)*prominence_frac
    cand = []
    for i in range(n):
        if y[i] > y[(i-1)%n] and y[i] >= y[(i+1)%n]:
            # local min left/right until higher point or half circumference
            lm = y[i]
            j=(i-1)%n
            steps=0
            left_min=y[j]
            while steps < n//2 and y[j] <= y[i]:
                left_min = min(left_min, y[j])
                j=(j-1)%n
                steps += 1
            j=(i+1)%n
            steps=0
            right_min=y[j]
            while steps < n//2 and y[j] <= y[i]:
                right_min = min(right_min, y[j])
                j=(j+1)%n
                steps += 1
            p = y[i] - max(left_min, right_min)
            if p >= prom:
                cand.append((i, y[i], p))
    # nonmax suppression by height
    cand = sorted(cand, key=lambda x:x[1], reverse=True)
    keep=[]
    for c in cand:
        if all(min((c[0]-k[0])%n, (k[0]-c[0])%n) >= min_sep for k in keep):
            keep.append(c)
    keep=sorted(keep)
    return keep

def radial_profile(xy, center, nbins=1440, stat='q90'):
    z = xy - center
    th = np.mod(np.arctan2(z[:,1], z[:,0]), 2*np.pi)
    r = np.hypot(z[:,0], z[:,1])
    bins = np.floor(th/(2*np.pi)*nbins).astype(int)
    prof = np.full(nbins, np.nan)
    counts = np.bincount(bins, minlength=nbins)
    for b in range(nbins):
        vals = r[bins==b]
        if len(vals):
            if stat == 'max': prof[b] = np.max(vals)
            elif stat == 'mean': prof[b] = np.mean(vals)
            else: prof[b] = np.quantile(vals, 0.90)
    # fill empty circular by interpolation
    if np.any(np.isnan(prof)):
        idx = np.arange(nbins)
        good = ~np.isnan(prof)
        # periodic interp
        good_idx = idx[good]
        good_val = prof[good]
        idx_ext = np.r_[good_idx-nbins, good_idx, good_idx+nbins]
        val_ext = np.r_[good_val, good_val, good_val]
        prof = np.interp(idx, idx_ext, val_ext)
    return prof, counts

def dominant_fft(profile, maxk=30):
    y = profile - np.mean(profile)
    amps = np.abs(np.fft.rfft(y))
    amps[0]=0
    # normalize by median/noise
    ks = np.arange(len(amps))
    top = sorted([(k, amps[k]) for k in range(1, min(maxk+1, len(amps)))], key=lambda x:x[1], reverse=True)[:8]
    return top, amps

def time_radial_peak_angles(xy, center):
    z=xy-center
    th=np.mod(np.arctan2(z[:,1], z[:,0]),2*np.pi)
    r=np.hypot(z[:,0],z[:,1])
    # smooth time series
    win=max(3, len(r)//1000)
    if win%2==0: win+=1
    rs=np.convolve(r, np.ones(win)/win, mode='same')
    loc=np.where((rs[1:-1]>rs[:-2]) & (rs[1:-1]>=rs[2:]))[0]+1
    # keep prominent high maxima
    if len(loc)==0: return np.array([])
    q=np.quantile(rs,0.65)
    loc=loc[rs[loc]>=q]
    return th[loc]

def angular_cluster_est(angs, maxN=30):
    if len(angs)<5: return []

    top_fft,_=dominant_fft(hs, maxN)
    return len(peaks), [(round(p[0]*360/nb,1), round(p[1],1)) for p in peaks], top_fft[:5]
