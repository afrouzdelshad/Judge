def angle_fft_scores(xy, center, nb=1440, kmax=30):
    z=(xy[:,0]-center[0])+1j*(xy[:,1]-center[1])
    r=np.abs(z)
    th=np.mod(np.angle(z),2*np.pi)
    bins=np.linspace(0,2*np.pi,nb+1)
    centers=(bins[:-1]+bins[1:])/2
    inds=np.minimum(np.searchsorted(bins, th, side='right')-1, nb-1)
    sums=np.zeros(nb); cnt=np.zeros(nb)
    for i,rv in zip(inds,r):
        sums[i]+=rv; cnt[i]+=1
    vals=np.full(nb,np.nan)
    good=cnt>0
    vals[good]=sums[good]/cnt[good]
    if good.sum()<10:
        return None
    vals=np.interp(centers, centers[good], vals[good], period=2*np.pi)
    vals-=vals.mean()
    amps=np.abs(np.fft.rfft(vals))
    return amps[:kmax+1]

def tip_angles_scores(xy, center, kmax=30, q=0.75):
    z=(xy[:,0]-center[0])+1j*(xy[:,1]-center[1])
    r=np.abs(z)
    # local maxima
    idx=[]
    for i in range(1,len(r)-1):
        if r[i] >= r[i-1] and r[i] > r[i+1]:
            idx.append(i)
    idx=np.array(idx)
    if len(idx)==0: 
        return None, 0
    # keep strongest maxima
    idx=idx[r[idx] >= np.quantile(r[idx], q)]
    phi=np.angle(z[idx])
    scores=[]
    for k in range(kmax+1):
        if k==0: scores.append(0.0)
        else: scores.append(abs(np.mean(np.exp(1j*k*phi))))
    return np.array(scores), len(idx)

def high_radius_scores(xy, center, kmax=30, q=0.90):
    z=(xy[:,0]-center[0])+1j*(xy[:,1]-center[1])
    r=np.abs(z)
    phi=np.angle(z[r>=np.quantile(r,q)])
    return np.array([0.0]+[abs(np.mean(np.exp(1j*k*phi))) for k in range(1,kmax+1)])
