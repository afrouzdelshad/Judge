```python
import csv, ast
import numpy as np

path='/Users/anxiaodong/Documents/GitHub/Judge/task1data_shuffled.csv'
with open(path) as f:
    r=csv.reader(f)
    header=next(r)
    rows=[row for row in r]
labels=header[1:]
data={l:[] for l in labels}
for row in rows:
    for i,l in enumerate(labels):
        data[l].append(ast.literal_eval(row[i+1]))
for l in labels:
    data[l]=np.array(data[l])

# For each trajectory, find local maxima in radius (lobe tips), get their angles,
# then estimate number of distinct angular lobe directions via FFT of radius vs angle
res={}
for l in labels:
    xy=data[l]
    cx,cy=xy[:,0].mean(),xy[:,1].mean()
    dx=xy[:,0]-cx; dy=xy[:,1]-cy
    r=np.hypot(dx,dy)
    ang=np.arctan2(dy,dx)
    # local maxima of r
    idx=np.where((r[1:-1]>r[:-2])&(r[1:-1]>r[2:]))[0]+1
    peaks=r[idx]
    # keep significant peaks (lobe tips)
    thr=np.percentile(r,60)
    tipang=ang[idx[peaks>thr]]
    # cluster angles into lobes: histogram
    h,_=np.histogram(tipang,bins=72,range=(-np.pi,np.pi))
    # count peaks in histogram of tip-angles
    nz=(h>0).sum()
    res[l]=(len(idx), nz, tipang)

for l in labels:
    print(l, res[l][0], res[l][1])
```
