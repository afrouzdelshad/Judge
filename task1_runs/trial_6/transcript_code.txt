```python
import csv, ast, numpy as np

path = r"D:\Documents\GitHub\LLMJudge\task1_data.csv"
rows = list(csv.reader(open(path)))
hdr = rows[0]; labels = hdr[1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {lab: np.array([ast.literal_eval(r[i+1]) for r in rows[1:]]) for i,lab in enumerate(labels)}

for lab in labels:
    a=data[lab]
    x=a[:,0]-np.median(a[:,0]); y=a[:,1]-np.median(a[:,1])
    th=np.unwrap(np.arctan2(y,x))
    r=np.hypot(x,y)
    total=abs(th[-1]-th[0])/(2*np.pi)
    # fourier of r vs theta: interpolate onto uniform theta grid
    s = np.sign(th[-1]-th[0])
    thm = s*th
    # ensure monotone
    mono = np.maximum.accumulate(thm)
    grid=np.linspace(mono[0],mono[-1],4000)
    ri=np.interp(grid,mono,r)
    ri=ri-ri.mean()
    F=np.abs(np.fft.rfft(ri))
    freqs=np.fft.rfftfreq(len(grid), d=(grid[1]-grid[0])/(2*np.pi))  # cycles per radian*2pi = per revolution
    # candidate harmonics 1..15
    scores=[]
    for n in range(1,16):
        idx=np.argmin(abs(freqs-n))
        # local max around
        w=F[max(0,idx-2):idx+3].max()
        scores.append((n,w))
    scores.sort(key=lambda z:-z[1])
    print(lab, "rev=%.1f"%total, "top:", [(n,round(v)) for n,v in scores[:4]])
```
