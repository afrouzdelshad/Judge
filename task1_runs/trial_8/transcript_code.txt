    cx,cy=(x.max()+x.min())/2,(y.max()+y.min())/2
    r=np.hypot(x-cx,y-cy); th=np.arctan2(y-cy,x-cx)
    idx=[i for i in range(1,len(r)-1) if r[i]>r[i-1] and r[i]>=r[i+1]]
    a=np.array([th[i] for i in idx])
    d=np.diff(a); d=(d+np.pi)%(2*np.pi)-np.pi

    # cluster angles mod 2pi
    am=np.sort(a%(2*np.pi))
    gaps=np.diff(np.concatenate([am,[am[0]+2*np.pi]]))
    ncl=np.sum(gaps>0.15)
    print(lab,"nmax=%d"%len(idx),"med_dth=%.4f"%md,"N=2pi/dth=%.2f"%(2*np.pi/abs(md)),"clusters=%d"%ncl,
