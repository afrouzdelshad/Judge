import csv, ast, numpy as np
path=r"D:\Documents\GitHub\LLMJudge\task1_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
data={l:np.array([ast.literal_eval(r[i+1]) for r in rows[1:]],dtype=float) for i,l in enumerate(labels)}
for l in labels:
    a=data[l]
    x=a[:,0]-a[:,0].mean(); y=a[:,1]-a[:,1].mean()
    th=np.arctan2(y,x); r=np.hypot(x,y)
    # harmonic amplitudes of r vs theta
    amps=[]
    for m in range(1,16):
        c=np.mean(r*np.exp(1j*m*th))
        amps.append(abs(c))
    amps=np.array(amps)
    best=np.argsort(amps)[::-1][:4]+1
    print(l, [f"{m}:{amps[m-1]:.3f}" for m in best])
