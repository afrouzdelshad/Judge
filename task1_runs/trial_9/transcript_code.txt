import csv, ast, numpy as np
path=r"D:\Documents\GitHub\LLMJudge\task1_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
t=np.array([float(r[0]) for r in rows[1:]])
dt=t[1]-t[0]
data={lab:np.array([ast.literal_eval(r[i+1]) for r in rows[1:]]) for i,lab in enumerate(labels)}
res={}
for lab in labels:
    a=data[lab]; x,y=a[:,0],a[:,1]
    cx,cy=x.mean(),y.mean()
    r=np.hypot(x-cx,y-cy); th=np.unwrap(np.arctan2(y-cy,x-cx))
    # FFT of r
    rr=r-r.mean(); w=np.hanning(len(rr))
    F=np.abs(np.fft.rfft(rr*w)); f=np.fft.rfftfreq(len(rr),dt)
    fr=f[np.argmax(F)]
    # angular freq: mean dtheta/dt
    ft=(th[-1]-th[0])/(t[-1]-t[0])/(2*np.pi)
    # also fft of theta residual
    thr=th-(th[0]+ft*2*np.pi*(t-t[0]))
    Ft=np.abs(np.fft.rfft(thr*w)); fth=f[np.argmax(Ft)]
    # angular positions of radial maxima
    idx=[i for i in range(1,len(r)-1) if r[i]>r[i-1] and r[i]>r[i+1]]
    dth=np.diff(th[idx])
    res[lab]=(fr,ft,fr/ft if ft!=0 else np.nan, fth, np.median(dth), 2*np.pi/np.median(dth) if np.median(dth)!=0 else 0)
    print(lab, f"fr={fr:.4f} fth={ft:.4f} N=fr/fth={fr/ft:.3f} fth_osc={fth:.4f} med_dtheta={np.median(dth):.4f} 2pi/dth={2*np.pi/np.median(dth):.2f} dth_min={dth.min():.3f} dth_max={dth.max():.3f}")
