def circ_profile(theta, val=None, B=720):
    u = theta/(2*np.pi)*B
    i = np.floor(u).astype(int) % B
    if val is None:
        return np.bincount(i, minlength=B).astype(float)
    cnt = np.bincount(i, minlength=B)
    sm = np.bincount(i, weights=val, minlength=B)
    good = cnt > 0
    a = np.divide(sm, cnt, out=np.full(B, np.nan), where=good)
    if not good.all():
        g = np.flatnonzero(good)
        a = np.interp(np.arange(B), np.r_[g-B,g,g+B], np.r_[a[g],a[g],a[g]])
    return a

def smooth_circular(a, sigma):
    n=len(a)
    k=np.fft.fftfreq(n)*n
    return np.fft.ifft(np.fft.fft(a)*np.exp(-.5*(2*np.pi*sigma*k/n)**2)).real

def peaks_circular(a, minprom=.08):
    b=(a>np.roll(a,1))&(a>=np.roll(a,-1))
    inds=np.flatnonzero(b)
    # retain maxima substantially above neighboring angular valleys
    keep=[]
    n=len(a)
    w=max(4,n//50)
    for i in inds:
        local=np.r_[a[(np.arange(i-w,i)%n)],a[(np.arange(i+1,i+w+1)%n)]]
        if a[i]-np.min(local) >= minprom*np.ptp(a):
            keep.append(i)
    return keep

def kmeans(X,k,seed=1,its=60):

        d=np.min(((X[:,None,:]-np.array(C)[None,:,:])**2).sum(2),axis=1)
        C.append(X[rng.choice(len(X),p=d/d.sum())])
    C=np.array(C)
    for _ in range(its):
        ds=((X[:,None,:]-C[None,:,:])**2).sum(2)
