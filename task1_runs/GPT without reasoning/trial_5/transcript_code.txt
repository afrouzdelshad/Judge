    centers = []
    centers.append(("mean", np.mean(p, axis=0)))
    centers.append(("median", np.median(p, axis=0)))

    # Algebraic least-squares circle center.
    A = np.c_[2*x, 2*y, np.ones(len(x))]
    b = x*x + y*y
    sol, *_ = np.linalg.lstsq(A, b, rcond=None)
    centers.append(("circle", sol[:2]))

    summaries = []
    for cname, c in centers:
        q = p - c
        r = np.hypot(q[:,0], q[:,1])
        th = np.unwrap(np.arctan2(q[:,1], q[:,0]))
        turns = abs((th[-1] - th[0]) / (2*np.pi))

        # Fit radial variation as Fourier modes in geometric polar angle.
        # Scores are normalized amplitudes for modes 1..30.
        scores = []
        for n in range(1, 31):
            M = np.c_[np.ones(len(th)), np.cos(n*th), np.sin(n*th)]
            coef, *_ = np.linalg.lstsq(M, r, rcond=None)
            scores.append(np.hypot(coef[1], coef[2]) / (np.std(r) + 1e-12))
        top = np.argsort(scores)[::-1][:6] + 1

        # Angular-bin average radius and count its smoothed maxima.
        phi = np.mod(np.arctan2(q[:,1], q[:,0]), 2*np.pi)
        nb = 720
        idx = np.floor(phi/(2*np.pi)*nb).astype(int) % nb
        sums = np.bincount(idx, weights=r, minlength=nb)
        cnts = np.bincount(idx, minlength=nb)
        rb = np.divide(sums, cnts, out=np.full(nb, np.nan), where=cnts>0)
        good = np.isfinite(rb)
        if not np.all(good):
            inds = np.arange(nb)
            ext_i = np.r_[inds[good]-nb, inds[good], inds[good]+nb]
            ext_v = np.r_[rb[good], rb[good], rb[good]]
            rb = np.interp(inds, ext_i, ext_v)
        sm = gaussian_filter1d(rb, 5, mode="wrap")
        prom = max(0.08*(np.max(sm)-np.min(sm)), 1e-12)
        pk, _ = find_peaks(np.r_[sm, sm, sm], prominence=prom,
                           distance=max(3, nb//60))
        pk = pk[(pk >= nb) & (pk < 2*nb)] - nb

        summaries.append(
            (cname, np.round(c,3), round(turns,2), len(pk),
             [(int(n), round(scores[n-1],3)) for n in top])
        )

    # Temporal radial spectrum using mean center, useful if angular speed varies.
    c = np.mean(p, axis=0)
    rr = np.hypot(x-c[0], y-c[1])
    rr = rr - np.mean(rr)

    temp_top = [(round(freqs[i],5), round(spec[i]/(spec[ids[0]]+1e-12),3))
                for i in ids]

    print("\n", lab, "range", np.round(np.ptp(p,axis=0),3),
          "temporal", temp_top)
    for s in summaries:
        print(" ", s)

def circ_smooth(a, sigma):
    n = len(a)
    k = np.minimum(np.arange(n), n-np.arange(n))
    ker = np.exp(-0.5*(k/sigma)**2)
    ker /= ker.sum()
    return np.fft.ifft(np.fft.fft(a)*np.fft.fft(ker)).real

def periodic_peaks(a, minprom=0.05):
    prev, nxt = np.roll(a,1), np.roll(a,-1)
    cand = np.where((a > prev) & (a >= nxt))[0]
    rng = np.ptp(a)
    out = []
    n = len(a)
    for i in cand:
        # local prominence against minima within roughly 1/12 circle
        w = max(5, n//30)
        left = np.min(a[(i-np.arange(1,w+1))%n])
        right = np.min(a[(i+np.arange(1,w+1))%n])
        if a[i] - max(left,right) >= minprom*rng:
            out.append(i)
    return out
