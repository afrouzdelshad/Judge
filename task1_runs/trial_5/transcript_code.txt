import csv, ast, numpy as np
path=r"D:\Documents\GitHub\LLMJudge\task1_data.csv"
rows=list(csv.reader(open(path)))
labels=rows[0][1:]
data={}
for i,l in enumerate(labels):
    data[l]=np.array([ast.literal_eval(r[i+1]) for r in rows[1:]])

for l in labels:
    p=data[l]; x,y=p[:,0],p[:,1]
    r=np.hypot(x,y); th=np.arctan2(y,x)%(2*np.pi)
    idx=[i for i in range(1,len(r)-1) if r[i]>r[i-1] and r[i]>=r[i+1]]
    ang=np.sort(th[idx])
    # cluster angles on circle with gap threshold
    gaps=np.diff(np.concatenate([ang,[ang[0]+2*np.pi]]))
    # try thresholds
    res={}
    for thr in [0.05,0.1,0.15,0.2,0.3]:
        res[thr]=int(np.sum(gaps>thr))
    # also histogram-based
    print(l, res, "n_max=%d"%len(idx))
