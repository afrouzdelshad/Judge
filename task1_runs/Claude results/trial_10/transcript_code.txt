import csv, ast, numpy as np
path = r"D:\Documents\GitHub\LLMJudge\task1_data.csv"
with open(path) as f:
    r=csv.reader(f); header=next(r); rows=list(r)
labels=header[1:]
t=np.array([float(x[0]) for x in rows])
for lab_i,lab in enumerate(labels):
    p=np.array([ast.literal_eval(row[lab_i+1]) for row in rows])
    x,y=p[:,0]-p[:,0].mean(),p[:,1]-p[:,1].mean()
    rr=np.hypot(x,y); th=np.arctan2(y,x)
    idx=np.where((rr[1:-1]>rr[:-2])&(rr[1:-1]>=rr[2:]))[0]+1
    # keep only prominent maxima (r above median)
    idx=idx[rr[idx]>np.percentile(rr,60)]
    ang=np.mod(th[idx],2*np.pi)
    # find best N: minimize spread of ang mod 2pi/N
    best=[]
    for N in range(2,25):
        a=np.mod(ang, 2*np.pi/N)*N  # scaled to [0,2pi)
        # circular concentration
        R=abs(np.mean(np.exp(1j*a)))
        best.append((R,N))
    best.sort(reverse=True)
    print(lab, "nmax=%d"%len(idx), [(N,round(R,3)) for R,N in best[:4]])
