import csv, ast, numpy as np

path = r"D:\Documents\GitHub\LLMJudge\task1_data.csv"
rows = list(csv.reader(open(path)))
hdr = rows[0]
labels = hdr[1:]
t = []
data = {l: [] for l in labels}
for r in rows[1:]:
    t.append(float(r[0]))
    for l, c in zip(labels, r[1:]):
        x, y = ast.literal_eval(c)
        data[l].append((x, y))
t = np.array(t)
print(len(t), t[0], t[-1], labels)

for l in labels:
    a = np.array(data[l])
    x, y = a[:,0], a[:,1]
    cx, cy = x.mean(), y.mean()
    X, Y = x-cx, y-cy
    th = np.unwrap(np.arctan2(Y, X))
    total_turn = abs(th[-1]-th[0])/(2*np.pi)
    r = np.hypot(X, Y)
    # count local maxima of r
    m = (r[1:-1] > r[:-2]) & (r[1:-1] > r[2:])
    nmax = m.sum()
    # smoothed version
    k = 5
    rs = np.convolve(r, np.ones(k)/k, mode='same')
    ms = (rs[1:-1] > rs[:-2]) & (rs[1:-1] > rs[2:])
    print(l, "turns=%.2f"%total_turn, "nmax=",nmax, "nmax_s=",ms.sum(),
          "ratio=%.2f"%(nmax/total_turn), "ratio_s=%.2f"%(ms.sum()/total_turn),
          "r_mean=%.2f"%r.mean(), "r_std=%.2f"%r.std())

    a=np.array(data[l]); x,y=a[:,0]-a[:,0].mean(),a[:,1]-a[:,1].mean()
    r=np.hypot(x,y); th=np.arctan2(y,x)
    idx=np.where((r[1:-1]>r[:-2])&(r[1:-1]>r[2:]))[0]+1
    # keep only prominent maxima (tips)

    tips=th[idx]
    scores=[(abs(np.mean(np.exp(1j*n*tips))),n) for n in range(1,80)]
