with open(path, newline="") as f:

for lab in labels:
    xy = data[lab]
    z = xy[:, 0] + 1j * xy[:, 1]

    # Try both mean and bounding-box centers.
    centers = [
        np.mean(z),
        (xy[:,0].min()+xy[:,0].max())/2 + 1j*(xy[:,1].min()+xy[:,1].max())/2
    ]

    reports = []
    for center in centers:
        zc = z - center
        theta = np.unwrap(np.angle(zc))
        direction = 1 if np.polyfit(t, theta, 1)[0] >= 0 else -1
        phi = direction * theta

        # Harmonic least-squares score of radius versus angular phase.
        scores = []
        amplitudes = []

            c, s = np.cos(k*phi), np.sin(k*phi)
            X = np.column_stack((np.ones(len(phi)), c, s))
            coef = np.linalg.lstsq(X, radius, rcond=None)[0]
            residual = radius - X.dot(coef)
            scores.append(1 - np.var(residual)/np.var(radius))
            amplitudes.append(np.hypot(coef[1], coef[2]))

        nb = 1440

        sums = np.bincount(bins, weights=radius, minlength=nb)
