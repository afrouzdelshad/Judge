import csv, ast, numpy as np

path = "/Users/adelshad3/Documents/GitHub/Judge/task1_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {lab: np.array([ast.literal_eval(r[j+1]) for r in rows[1:]], float)
        for j, lab in enumerate(labels)}

print("nrows", len(t), "labels", labels)
for lab in labels:
    z = data[lab]
    x, y = z[:,0], z[:,1]
    # Center robustly, then polar coordinates
    cx, cy = np.mean(x), np.mean(y)
    X, Y = x-cx, y-cy
    r = np.hypot(X,Y)
    th = np.unwrap(np.arctan2(Y,X))
    turns = (th[-1]-th[0])/(2*np.pi)
    
    # Correlation/spectral strength of radius as function of angle for candidate lobe counts.
    # Fit r = const + harmonics sin(N theta), cos(N theta), score normalized amplitude.
    scores = []
    for n in range(1, 41):
        A = np.column_stack([np.ones(len(r)), np.cos(n*th), np.sin(n*th)])
        coef = np.linalg.lstsq(A, r, rcond=None)[0]
        amp = np.hypot(coef[1],coef[2])
        scores.append(amp/np.std(r) if np.std(r)>0 else 0)
    tops = np.argsort(scores)[::-1][:8] + 1
    
    # Count radial maxima with modest prominence using scipy
    try:
        from scipy.signal import find_peaks
        prom = max(np.std(r)*0.25, 1e-12)
        peaks,_ = find_peaks(r, prominence=prom, distance=max(2,len(r)//500))
        peak_count = len(peaks)
    except Exception:
        peak_count = -1
        
    print(lab, "turns", round(turns,3), "center", (round(cx,3),round(cy,3)),
          "rmean/std", (round(r.mean(),3),round(r.std(),3)),
          "topN", [(int(n),round(scores[n-1],3)) for n in tops],
          "timepeaks", peak_count)
