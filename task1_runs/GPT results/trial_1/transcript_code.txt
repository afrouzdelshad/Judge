import csv, ast, numpy as np
from scipy.signal import find_peaks, periodogram
from scipy.cluster.vq import kmeans2

path = "/Users/adelshad3/Documents/GitHub/Judge/task1_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))
labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {}
for j, lab in enumerate(labels, 1):
    data[lab] = np.array([ast.literal_eval(r[j]) for r in rows[1:]], float)

print("labels", labels, "nrows", len(t), "time", (t[0], t[-1]), "dt", np.median(np.diff(t)))
print("lab  xrange yrange  windings  theta_monotonicity  radial_peak_count  spectral_ratio  top_angle_harmonics")
for lab in labels:
    xy = data[lab]
    # center robustly; compare mean and bounding-box center
    c = np.mean(xy, axis=0)
    z = (xy[:,0]-c[0]) + 1j*(xy[:,1]-c[1])
    th = np.unwrap(np.angle(z))
    r = np.abs(z)
    wind = (th[-1]-th[0])/(2*np.pi)
    mono = abs(np.mean(np.sign(np.diff(th))))
    peaks,_ = find_peaks(r, prominence=.08*np.ptp(r), distance=max(2,len(r)//1000))
    # dominant temporal frequencies of angle velocity and radius
    rv = r-r.mean()
    f,p = periodogram(rv)
    inds=np.argsort(p[1:])[-5:]+1
    # angular Fourier coefficients of radial shape: exp(-ik theta)
    harms=[]
    for k in range(1,31):
        harms.append(abs(np.mean((rv/(np.std(rv)+1e-12))*np.exp(-1j*k*th))))
    top=np.argsort(harms)[-8:][::-1]+1
    # ratio dominant radial cycles / absolute windings
    dom=f[inds[-1]]*(t[-1]-t[0])
    ratio=dom/(abs(wind)+1e-12)
    print(f"{lab:>2} {np.ptp(xy[:,0]):7.2f} {np.ptp(xy[:,1]):7.2f} {wind:9.2f} {mono:7.3f} {len(peaks):6d} {ratio:9.2f} "
          + " ".join(f"{k}:{harms[k-1]:.2f}" for k in top[:5]))

# Also estimate number of spatial density clusters by silhouette for k 2..21 on subsample.
from scipy.cluster.vq import kmeans2
from scipy.spatial.distance import cdist
rng=np.random.default_rng(0)

    X=X[np.linspace(0,len(X)-1,min(2500,len(X))).astype(int)]
    ds=[]
    for k in range(1,22):
        cen,idx=kmeans2(X,k,minit='++',iter=30,seed=k)
        ds.append(np.mean(np.sum((X-cen[idx])**2,axis=1)))
    imp=np.array(ds[:-1])/np.array(ds[1:])
    tops=np.argsort(imp)[-5:][::-1]+2
    print(lab, " ".join(f"{k}:{imp[k-2]:.2f}" for k in tops), "d:", " ".join(f"{v:.3f}" for v in ds[:10]))
