import csv, ast, numpy as np
from scipy.signal import find_peaks
from scipy.fft import rfft, rfftfreq

path = "/Users/adelshad3/Documents/GitHub/Judge/task1_data.csv"
with open(path, newline="") as f:
    rows = list(csv.reader(f))

labels = rows[0][1:]
t = np.array([float(r[0]) for r in rows[1:]])
data = {}
for j, lab in enumerate(labels, 1):
    data[lab] = np.array([ast.literal_eval(r[j]) for r in rows[1:]], float)

print("shape", len(t), labels, "dt", np.median(np.diff(t)))
for lab in labels:
    z = data[lab][:,0] + 1j*data[lab][:,1]
    # Estimate center robustly and inspect radial oscillations against unwrapped angle.
    c = np.mean(z)
    zz = z-c
    th = np.unwrap(np.angle(zz))
    r = np.abs(zz)
    turns = abs((th[-1]-th[0])/(2*np.pi))
    # Count radial maxima with several prominence levels.
    counts = []
    for p in [0.05, .1, .2, .3, .4]:
        pk,_=find_peaks(r, prominence=p*np.ptp(r), distance=max(1,len(r)//500))
        counts.append(len(pk))
    # Frequency ratios: dominant radial temporal frequency / angular turn frequency.
    rr=r-np.mean(r)
    spec=np.abs(rfft(rr))
    freqs=rfftfreq(len(r), np.median(np.diff(t)))
    inds=np.argsort(spec[1:])[-8:]+1
    top=sorted([(spec[i],freqs[i]) for i in inds], reverse=True)[:5]
    turnrate=turns/(t[-1]-t[0])
    ratios=[round(f/turnrate,3) if turnrate else None for _,f in top]
    # occupancy / Fourier harmonics of shape sampled by angle: fit r to k theta.
    harms=[]
    for k in range(1,31):
        A=np.column_stack([np.ones(len(r)),np.cos(k*th),np.sin(k*th)])
        coef=np.linalg.lstsq(A,r,rcond=None)[0]
        harms.append(np.hypot(coef[1],coef[2]))
    best=np.argsort(harms)[::-1][:6]+1
    print(lab, "center",np.round([c.real,c.imag],2),"turns",round(turns,2),
          "r-range",np.round([r.min(),r.max()],2),"peaks",counts,
          "freqrat",ratios,"harmbest",[(int(k),round(harms[k-1],3)) for k in best])
