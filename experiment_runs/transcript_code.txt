def est_counts(x,y,z):
    # center in xy and compute angle
    xc = x - np.mean(x)
    yc = y - np.mean(y)
    theta = np.arctan2(yc, xc)
    r = np.sqrt(xc*xc + yc*yc)
    # focus on outer-wing visits
    q = np.quantile(r, 0.7)
    th = theta[r >= q]
    # histogram peaks in angle
    bins = 720
    hist, edges = np.histogram(th, bins=bins, range=(-np.pi, np.pi))
    # smooth circularly
    k = np.array([1,2,3,2,1], float)
    pad = len(k)//2
    hh = np.r_[hist[-pad:], hist, hist[:pad]]
    sm = np.convolve(hh, k/k.sum(), mode='same')[pad:-pad]
    # local maxima with prominence-ish threshold
    mx = (sm > np.roll(sm,1)) & (sm >= np.roll(sm,-1))
    idx = np.where(mx)[0]
    vals = sm[idx]
    if len(vals)==0:
        return {}
    # count peaks above several thresholds
    out={}
    for frac in [0.15,0.2,0.25,0.3,0.35,0.4,0.5]:
        keep = vals >= frac*np.max(sm)
        # merge nearby peaks (within 10 bins circularly)
        sel = np.sort(idx[keep])
        if len(sel)==0:
            out[frac]=0
            continue
        groups=[[sel[0]]]
        for a in sel[1:]:
            if a-groups[-1][-1] <= 10:
                groups[-1].append(a)
            else:
                groups.append([a])
        if len(groups) > 1 and (groups[0][0] + bins - groups[-1][-1] <= 10):
            groups[0] = groups[-1] + groups[0]
            groups = groups[:-1]
        out[frac]=len(groups)
    return out, (np.mean(x), np.mean(y), np.mean(z), np.mean(r), np.std(r), np.min(r), np.max(r))
